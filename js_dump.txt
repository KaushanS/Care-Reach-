function openPatientModal(name, nic, dob, gender, phone, address, gName, gContact, relation) {
            document.getElementById('pmName').value = name;
            document.getElementById('pmNicLabel').value = nic;
            document.getElementById('pmDob').value = dob;
            document.getElementById('pmGender').value = gender;
            document.getElementById('pmPhone').value = phone;
            document.getElementById('pmAddress').value = address;
            document.getElementById('pmGuardian').value = gName;
            document.getElementById('pmGContact').value = gContact;
            document.getElementById('pmRelation').value = relation;
            document.getElementById('pmUsername').value = name;
            document.getElementById('pmPassword').value = nic;
            document.getElementById('patientModal').classList.add('active');
        }

        function openGuardianModal(name, phone, registerId) {
            document.getElementById('gmName').value = name;
            document.getElementById('gmPhone').value = phone;
            document.getElementById('gmUsername').value = name;
            document.getElementById('gmPassword').value = registerId;
            document.getElementById('guardianModal').classList.add('active');
        }

        function closeModal(id) {
            document.getElementById(id).classList.remove('active');
        }

        function submitModal(id, type) {
            alert(type + ' account created successfully!');
            closeModal(id);
        }
